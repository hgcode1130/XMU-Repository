package cn.edu.xmu.oomall.service.adapter.feign.fallback;

import cn.edu.xmu.javaee.core.model.ReturnNo;
import cn.edu.xmu.javaee.core.model.ReturnObject;
import cn.edu.xmu.oomall.service.adapter.feign.LogisticsService;
import cn.edu.xmu.oomall.service.adapter.feign.dto.CreateWaybillRequest;
import cn.edu.xmu.oomall.service.adapter.feign.dto.WaybillResponse;
import lombok.extern.slf4j.Slf4j;
import org.springframework.stereotype.Component;

import java.time.LocalDateTime;
import java.time.format.DateTimeFormatter;

/**
 * 物流服务降级处理
 * 当 logistics-service 不可用时，使用模拟数据继续业务流程
 */
@Component
@Slf4j
public class LogisticsServiceFallback implements LogisticsService {
    
    private static final DateTimeFormatter FORMATTER = DateTimeFormatter.ofPattern("yyyyMMddHHmmss");
    
    @Override
    public ReturnObject createWaybill(CreateWaybillRequest request) {
        // 生成模拟运单号
        String mockWaybillNo = "MOCK-" + LocalDateTime.now().format(FORMATTER);
        
        log.warn("⚠️ 物流服务不可用，使用降级处理 | 生成模拟运单号={}", mockWaybillNo);
        log.info("📦 模拟运单信息 | 寄件人={} 电话={} 地址={}", 
                request.getSenderName(), 
                request.getSenderPhone(), 
                request.getSenderAddress());
        
        // 返回模拟运单响应
        WaybillResponse mockResponse = WaybillResponse.builder()
                .waybillNo(mockWaybillNo)
                .status(0) // 待揽收
                .build();
        
        return new ReturnObject(ReturnNo.OK, "模拟创建运单成功（降级处理）", mockResponse);
    }
    
    @Override
    public ReturnObject cancelWaybill(String waybillNo) {
        log.warn("⚠️ 物流服务不可用，使用降级处理 | 模拟取消运单={}", waybillNo);
        log.info("📦 模拟取消运单成功 | waybillNo={}", waybillNo);
        
        return new ReturnObject(ReturnNo.OK, "模拟取消运单成功（降级处理）");
    }
}

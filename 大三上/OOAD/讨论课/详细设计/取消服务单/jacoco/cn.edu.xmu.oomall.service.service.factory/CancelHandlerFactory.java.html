package cn.edu.xmu.oomall.service.service.factory;

import cn.edu.xmu.javaee.core.exception.BusinessException;
import cn.edu.xmu.javaee.core.model.ReturnNo;
import cn.edu.xmu.oomall.service.service.handler.CancelHandler;
import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.stereotype.Component;

import java.util.List;

/**
 * 取消服务单处理器工厂
 * 根据服务类型和状态选择合适的处理器
 */
@Component
@RequiredArgsConstructor
@Slf4j
public class CancelHandlerFactory {
    
    private final List<CancelHandler> handlers;
    
    /**
     * 获取取消处理器
     * 
     * @param serviceType 服务类型
     * @param progress 服务单状态
     * @return 对应的处理器
     * @throws BusinessException 如果不支持该场景
     */
    public CancelHandler getCancelHandler(Integer serviceType, Integer progress) {
        log.debug("查找取消处理器 | serviceType={} progress={}", serviceType, progress);
        
        for (CancelHandler handler : handlers) {
            if (handler.supports(serviceType, progress)) {
                log.info("找到匹配的取消处理器 | handler={}", handler.getClass().getSimpleName());
                return handler;
            }
        }
        
        String message = String.format("不支持的取消场景: serviceType=%d, progress=%d", serviceType, progress);
        log.error(message);
        throw new BusinessException(ReturnNo.STATENOTALLOW, message);
    }
}

package cn.edu.xmu.oomall.service.service.factory;

import cn.edu.xmu.javaee.core.exception.BusinessException;
import cn.edu.xmu.javaee.core.model.ReturnNo;
import cn.edu.xmu.oomall.service.service.handler.AcceptHandler;
import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.stereotype.Component;

import java.util.List;

/**
 * 接受服务单处理器工厂
 * 根据服务类型选择合适的处理器
 */
@Component
@RequiredArgsConstructor
@Slf4j
public class AcceptHandlerFactory {
    
    private final List<AcceptHandler> handlers;
    
    /**
     * 获取接受处理器
     * 
     * @param serviceType 服务类型
     * @return 对应的处理器
     * @throws BusinessException 如果不支持该服务类型
     */
    public AcceptHandler getAcceptHandler(Integer serviceType) {
        log.debug("查找接受处理器 | serviceType={}", serviceType);
        
        for (AcceptHandler handler : handlers) {
            if (handler.supports(serviceType)) {
                log.info("找到匹配的接受处理器 | handler={}", handler.getClass().getSimpleName());
                return handler;
            }
        }
        
        String message = String.format("不支持的服务类型: serviceType=%d", serviceType);
        log.error(message);
        throw new BusinessException(ReturnNo.FIELD_NOTVALID, message);
    }
}

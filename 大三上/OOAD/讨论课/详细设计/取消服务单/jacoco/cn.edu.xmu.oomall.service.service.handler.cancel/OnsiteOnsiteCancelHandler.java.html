package cn.edu.xmu.oomall.service.service.handler.cancel;

import cn.edu.xmu.oomall.service.dao.bo.ServiceOrder;
import cn.edu.xmu.oomall.service.service.handler.CancelHandler;
import cn.edu.xmu.oomall.service.service.result.CancelResult;
import lombok.extern.slf4j.Slf4j;
import org.springframework.stereotype.Component;

/**
 * 上门服务-已上门状态取消处理器
 * 场景：上门服务单已上门，需要释放维修人员并记录现场情况
 */
@Component
@Slf4j
public class OnsiteOnsiteCancelHandler implements CancelHandler {
    
    @Override
    public boolean supports(Integer serviceType, Integer progress) {
        return ServiceOrder.TYPE_ONSITE == serviceType && ServiceOrder.PROGRESS_ONSITE == progress;
    }
    
    @Override
    public CancelResult cancel(ServiceOrder serviceOrder, String cancelReason) {
        log.info("开始处理上门服务-已上门状态取消 | orderId={} technicianId={}", 
                serviceOrder.getId(), serviceOrder.getTechnicianId());
        
        try {
            // 释放维修人员
            if (serviceOrder.getTechnicianId() != null) {
                // TODO: 调用维修人员服务释放资源
                log.info("释放维修人员 | technicianId={}", serviceOrder.getTechnicianId());
            }
            
            // 记录现场情况
            String evidence = String.format("服务单取消 - 原因: %s, 取消时间: %s", 
                    cancelReason != null ? cancelReason : "未提供", 
                    java.time.LocalDateTime.now());
            serviceOrder.setEvidence(evidence);
            
            // 更新服务单状态
            serviceOrder.setProgress(ServiceOrder.PROGRESS_CANCELED);
            
            return CancelResult.success("取消成功，已释放维修人员并记录现场");
            
        } catch (Exception e) {
            log.warn("处理过程中发生异常，但继续取消服务单 | error={}", e.getMessage());
            serviceOrder.setProgress(ServiceOrder.PROGRESS_CANCELED);
            return CancelResult.success("取消成功（部分处理失败，请人工核实）");
        }
    }
}

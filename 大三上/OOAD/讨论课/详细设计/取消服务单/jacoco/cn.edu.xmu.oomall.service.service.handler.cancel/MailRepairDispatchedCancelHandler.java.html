package cn.edu.xmu.oomall.service.service.handler.cancel;

import cn.edu.xmu.oomall.service.adapter.feign.LogisticsService;
import cn.edu.xmu.oomall.service.dao.bo.ServiceOrder;
import cn.edu.xmu.oomall.service.service.handler.CancelHandler;
import cn.edu.xmu.oomall.service.service.result.CancelResult;
import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.stereotype.Component;

/**
 * 寄修服务-已派工状态取消处理器
 * 场景：寄修服务单已派工，需要释放维修人员并取消运单
 */
@Component
@RequiredArgsConstructor
@Slf4j
public class MailRepairDispatchedCancelHandler implements CancelHandler {
    
    private final LogisticsService logisticsService;
    
    @Override
    public boolean supports(Integer serviceType, Integer progress) {
        return ServiceOrder.TYPE_MAIL == serviceType && ServiceOrder.PROGRESS_DISPATCHED == progress;
    }
    
    @Override
    public CancelResult cancel(ServiceOrder serviceOrder, String cancelReason) {
        log.info("开始处理寄修服务-已派工状态取消 | orderId={} technicianId={}", 
                serviceOrder.getId(), serviceOrder.getTechnicianId());
        
        try {
            // 释放维修人员（这里简化处理，实际应该调用维修人员服务）
            if (serviceOrder.getTechnicianId() != null) {
                // TODO: 调用维修人员服务释放资源
                log.info("释放维修人员 | technicianId={}", serviceOrder.getTechnicianId());
            }
            
            // 取消运单
            if (serviceOrder.getWaybillNo() != null) {
                logisticsService.cancelWaybill(serviceOrder.getWaybillNo());
                log.info("成功取消运单 | waybillNo={}", serviceOrder.getWaybillNo());
            }
            
            // 更新服务单状态
            serviceOrder.setProgress(ServiceOrder.PROGRESS_CANCELED);
            
            return CancelResult.success("取消成功，已释放资源并取消运单");
            
        } catch (Exception e) {
            log.warn("取消过程中发生异常，但继续取消服务单 | error={}", e.getMessage());
            serviceOrder.setProgress(ServiceOrder.PROGRESS_CANCELED);
            return CancelResult.success("取消成功（部分资源释放失败，请人工处理）");
        }
    }
}

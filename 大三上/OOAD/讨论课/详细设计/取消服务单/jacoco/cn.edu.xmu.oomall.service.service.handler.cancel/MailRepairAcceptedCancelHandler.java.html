package cn.edu.xmu.oomall.service.service.handler.cancel;

import cn.edu.xmu.javaee.core.exception.BusinessException;
import cn.edu.xmu.javaee.core.model.ReturnNo;
import cn.edu.xmu.oomall.service.adapter.feign.LogisticsService;
import cn.edu.xmu.oomall.service.dao.bo.ServiceOrder;
import cn.edu.xmu.oomall.service.service.handler.CancelHandler;
import cn.edu.xmu.oomall.service.service.result.CancelResult;
import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.stereotype.Component;

/**
 * 寄修服务-已接受状态取消处理器
 * 场景：寄修服务单已被接受，需要取消运单
 */
@Component
@RequiredArgsConstructor
@Slf4j
public class MailRepairAcceptedCancelHandler implements CancelHandler {
    
    private final LogisticsService logisticsService;
    
    @Override
    public boolean supports(Integer serviceType, Integer progress) {
        return ServiceOrder.TYPE_MAIL == serviceType && ServiceOrder.PROGRESS_ACCEPTED == progress;
    }    
    @Override
    public CancelResult cancel(ServiceOrder serviceOrder, String cancelReason) {
        log.info("开始处理寄修服务-已接受状态取消 | orderId={} waybillNo={}", 
                serviceOrder.getId(), serviceOrder.getWaybillNo());
        
        try {
            // 取消运单
            if (serviceOrder.getWaybillNo() != null) {
                logisticsService.cancelWaybill(serviceOrder.getWaybillNo());
                log.info("成功取消运单 | waybillNo={}", serviceOrder.getWaybillNo());
            }
            
            // 更新服务单状态
            serviceOrder.setProgress(ServiceOrder.PROGRESS_CANCELED);
            
            return CancelResult.success("取消成功，已取消运单");
            
        } catch (Exception e) {
            log.warn("取消运单失败，但继续取消服务单 | waybillNo={} error={}", 
                    serviceOrder.getWaybillNo(), e.getMessage());
            
            // 运单取消失败不阻断服务单取消流程
            serviceOrder.setProgress(ServiceOrder.PROGRESS_CANCELED);
            return CancelResult.success("取消成功（运单取消失败，请人工处理）");
        }
    }
}

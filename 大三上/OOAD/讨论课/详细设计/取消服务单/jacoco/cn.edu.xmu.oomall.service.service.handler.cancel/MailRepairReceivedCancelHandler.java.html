package cn.edu.xmu.oomall.service.service.handler.cancel;

import cn.edu.xmu.oomall.service.adapter.feign.LogisticsService;
import cn.edu.xmu.oomall.service.adapter.feign.dto.CreateWaybillRequest;
import cn.edu.xmu.oomall.service.adapter.feign.dto.WaybillResponse;
import cn.edu.xmu.oomall.service.dao.bo.ServiceOrder;
import cn.edu.xmu.oomall.service.service.handler.CancelHandler;
import cn.edu.xmu.oomall.service.service.result.CancelResult;
import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.stereotype.Component;

/**
 * 寄修服务-已收件状态取消处理器
 * 场景：寄修服务单已收件，需要生成退回运单
 */
@Component
@RequiredArgsConstructor
@Slf4j
public class MailRepairReceivedCancelHandler implements CancelHandler {
    
    private final LogisticsService logisticsService;
    
    @Override
    public boolean supports(Integer serviceType, Integer progress) {
        return ServiceOrder.TYPE_MAIL == serviceType && ServiceOrder.PROGRESS_RECEIVED == progress;
    }    
    @Override
    public CancelResult cancel(ServiceOrder serviceOrder, String cancelReason) {
        log.info("开始处理寄修服务-已收件状态取消 | orderId={}", serviceOrder.getId());
        
        try {
            // 创建退回运单
            CreateWaybillRequest request = CreateWaybillRequest.builder()
                    .receiverName(serviceOrder.getContact())
                    .receiverPhone(serviceOrder.getPhone())
                    .receiverRegionId(serviceOrder.getRegionId())
                    .receiverAddress(serviceOrder.getAddress())
                    .senderName("维修中心")
                    .senderPhone("400-000-0000")
                    .senderRegionId(1L) // 维修中心地区ID
                    .senderAddress("维修中心地址")
                    .build();
            
            var response = logisticsService.createWaybill(request);
            if (response.getData() != null) {
                WaybillResponse waybillResponse = (WaybillResponse) response.getData();
                String newWaybillNo = waybillResponse.getWaybillNo();
                
                // 更新服务单状态和运单号
                serviceOrder.setProgress(ServiceOrder.PROGRESS_CANCELED);
                serviceOrder.setWaybillNo(newWaybillNo);
                
                log.info("成功创建退回运单 | newWaybillNo={}", newWaybillNo);
                return CancelResult.success("取消成功，已安排退回", newWaybillNo);
            } else {
                throw new RuntimeException("创建退回运单失败");
            }
            
        } catch (Exception e) {
            log.error("创建退回运单失败 | error={}", e.getMessage(), e);
            throw new RuntimeException("无法创建退回运单，取消失败: " + e.getMessage());
        }
    }
}

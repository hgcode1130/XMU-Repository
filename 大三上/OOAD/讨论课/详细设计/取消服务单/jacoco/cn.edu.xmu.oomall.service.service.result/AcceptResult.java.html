package cn.edu.xmu.oomall.service.service.result;

import lombok.AllArgsConstructor;
import lombok.Builder;
import lombok.Data;
import lombok.NoArgsConstructor;

/**
 * 接受服务单操作结果
 */
@Data
@Builder
@NoArgsConstructor
@AllArgsConstructor
public class AcceptResult {
    
    /**
     * 操作是否成功
     */
    private boolean success;
    
    /**
     * 结果消息
     */
    private String message;
    
    /**
     * 生成的运单号（仅寄修场景）
     */
    private String waybillNo;
    
    /**
     * 创建成功结果
     */
    public static AcceptResult success(String message) {
        return AcceptResult.builder()
                .success(true)
                .message(message)
                .build();
    }
    
    /**
     * 创建成功结果（带运单号）
     */
    public static AcceptResult success(String message, String waybillNo) {
        return AcceptResult.builder()
                .success(true)
                .message(message)
                .waybillNo(waybillNo)
                .build();
    }
    
    /**
     * 创建失败结果
     */
    public static AcceptResult failure(String message) {
        return AcceptResult.builder()
                .success(false)
                .message(message)
                .build();
    }
}

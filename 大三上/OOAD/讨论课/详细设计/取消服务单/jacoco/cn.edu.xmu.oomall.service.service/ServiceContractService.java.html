package cn.edu.xmu.oomall.service.service;

import cn.edu.xmu.javaee.core.exception.BusinessException;
import cn.edu.xmu.javaee.core.model.ReturnNo;
import cn.edu.xmu.javaee.core.model.ReturnObject;
import cn.edu.xmu.javaee.core.model.UserToken;
import cn.edu.xmu.oomall.service.dao.ContractHistoryRepository;
import cn.edu.xmu.oomall.service.dao.ServiceContractRepository;
import cn.edu.xmu.oomall.service.dao.bo.ContractHistory;
import cn.edu.xmu.oomall.service.dao.bo.ServiceContract;
import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

/**
 * 服务合同用例服务
 */
@Service
@RequiredArgsConstructor
@Slf4j
public class ServiceContractService {

    private final ServiceContractRepository serviceContractRepository;
    private final ContractHistoryRepository contractHistoryRepository;

    @Transactional
    public ReturnObject cancel(Long did, Long contractId, String reason, UserToken user) {
        ServiceContract contract = serviceContractRepository.findById(contractId)
                .orElseThrow(() -> new BusinessException(ReturnNo.RESOURCE_ID_NOTEXIST,
                        String.format("服务合同不存在，id=%d", contractId)));
        ContractHistory history = contract.cancel(did, reason, user);
        serviceContractRepository.save(contract);
        contractHistoryRepository.save(history);
        log.info("provider cancel contract success | did={} contractId={}", did, contractId);
        return new ReturnObject(ReturnNo.OK, new CancelContractResult(contract, history.getReason()));
    }

    /**
     * 取消结果封装，便于 Controller 构造 VO
     */
    public record CancelContractResult(ServiceContract contract, String reason) {
    }
}

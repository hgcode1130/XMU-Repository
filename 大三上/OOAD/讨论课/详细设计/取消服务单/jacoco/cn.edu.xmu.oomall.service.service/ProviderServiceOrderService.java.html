package cn.edu.xmu.oomall.service.service;

import cn.edu.xmu.javaee.core.exception.BusinessException;
import cn.edu.xmu.javaee.core.model.ReturnNo;
import cn.edu.xmu.javaee.core.model.ReturnObject;
import cn.edu.xmu.javaee.core.model.UserToken;
import cn.edu.xmu.oomall.service.dao.ServiceContractRepository;
import cn.edu.xmu.oomall.service.dao.ServiceOrderHistoryRepository;
import cn.edu.xmu.oomall.service.dao.ServiceOrderRepository;
import cn.edu.xmu.oomall.service.dao.bo.ServiceContract;
import cn.edu.xmu.oomall.service.dao.bo.ServiceOrder;
import cn.edu.xmu.oomall.service.dao.bo.ServiceOrderHistory;
import cn.edu.xmu.oomall.service.service.factory.AcceptHandlerFactory;
import cn.edu.xmu.oomall.service.service.factory.CancelHandlerFactory;
import cn.edu.xmu.oomall.service.service.handler.AcceptHandler;
import cn.edu.xmu.oomall.service.service.handler.CancelHandler;
import cn.edu.xmu.oomall.service.service.result.AcceptResult;
import cn.edu.xmu.oomall.service.service.result.CancelResult;
import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

/**
 * 服务商侧服务单用例服务
 * 使用策略模式处理不同服务类型和状态的业务逻辑
 */
@Service
@RequiredArgsConstructor
@Slf4j
public class ProviderServiceOrderService {

    private final ServiceOrderRepository serviceOrderRepository;
    private final ServiceOrderHistoryRepository serviceOrderHistoryRepository;
    private final ServiceContractRepository serviceContractRepository;
    private final CancelHandlerFactory cancelHandlerFactory;
    private final AcceptHandlerFactory acceptHandlerFactory;

    /**
     * 取消服务单（使用策略模式）
     * 
     * @param did 服务商ID
     * @param orderId 服务单ID
     * @param result 取消原因
     * @param user 操作用户
     * @return 操作结果
     */
    @Transactional
    public ReturnObject cancel(Long did, Long orderId, String result, UserToken user) {
        log.info("开始取消服务单 | did={} orderId={} result={}", did, orderId, result);
        
        // 1. 加载服务单并校验权限
        ServiceOrder order = loadAndValidateServiceOrder(orderId, did);
        
        // 2. 校验状态是否允许取消
        if (!canCancel(order.getProgress())) {
            throw new BusinessException(ReturnNo.STATENOTALLOW,
                    String.format("服务单(id=%d)状态%d不允许取消", orderId, order.getProgress()));
        }
        
        // 3. 获取对应的取消处理器
        CancelHandler handler = cancelHandlerFactory.getCancelHandler(order.getType(), order.getProgress());
        
        // 4. 记录原状态
        Integer lastStatus = order.getProgress();
        
        // 5. 执行取消逻辑
        CancelResult cancelResult = handler.cancel(order, result);
        
        // 6. 保存服务单变更
        serviceOrderRepository.save(order);
        
        // 7. 记录操作历史
        ServiceOrderHistory history = buildCancelHistory(order, lastStatus, result, did, user);
        serviceOrderHistoryRepository.save(history);
        
        log.info("取消服务单成功 | did={} orderId={} result={}", did, orderId, cancelResult.getMessage());
        return new ReturnObject(ReturnNo.OK);
    }
    
    /**
     * 接受服务单（使用策略模式）
     * 
     * @param did 服务商ID
     * @param orderId 服务单ID
     * @param user 操作用户
     * @return 操作结果
     */
    @Transactional
    public ReturnObject accept(Long did, Long orderId, UserToken user) {
        log.info("开始接受服务单 | did={} orderId={}", did, orderId);
        
        // 1. 加载服务单并校验权限
        ServiceOrder order = loadAndValidateServiceOrder(orderId, did);
        
        // 2. 校验状态是否为待接受
        if (order.getProgress() == null || order.getProgress() != ServiceOrder.PROGRESS_WAIT_ACCEPT) {
            throw new BusinessException(ReturnNo.STATENOTALLOW,
                    String.format("服务单(id=%d)状态%d不是待接受状态", orderId, order.getProgress()));
        }
        
        // 3. 获取对应的接受处理器
        AcceptHandler handler = acceptHandlerFactory.getAcceptHandler(order.getType());
        
        // 4. 执行接受逻辑
        AcceptResult acceptResult = handler.accept(order, did);
        
        // 5. 保存服务单变更
        serviceOrderRepository.save(order);
        
        // 6. 记录操作历史
        ServiceOrderHistory history = buildAcceptHistory(order, did, user);
        serviceOrderHistoryRepository.save(history);
        
        log.info("接受服务单成功 | did={} orderId={} result={}", did, orderId, acceptResult.getMessage());
        return new ReturnObject(ReturnNo.OK);
    }

    
    /**
     * 加载服务单并校验权限
     */
    private ServiceOrder loadAndValidateServiceOrder(Long orderId, Long did) {
        ServiceOrder order = serviceOrderRepository.findById(orderId)
                .orElseThrow(() -> new BusinessException(ReturnNo.RESOURCE_ID_NOTEXIST,
                        String.format("服务单不存在，id=%d", orderId)));
        
        if (order.getServiceContractId() == null) {
            throw new BusinessException(ReturnNo.RESOURCE_ID_OUTSCOPE,
                    String.format("服务单(id=%d)未绑定服务合同，无法校验服务商", orderId));
        }
        
        ServiceContract contract = serviceContractRepository.findById(order.getServiceContractId())
                .orElseThrow(() -> new BusinessException(ReturnNo.RESOURCE_ID_OUTSCOPE,
                        String.format("服务单(id=%d)未找到对应合同(%d)", orderId, order.getServiceContractId())));
        
        if (!contract.ownedBy(did)) {
            throw new BusinessException(ReturnNo.RESOURCE_ID_OUTSCOPE,
                    String.format("服务单(id=%d)不属于服务商(%d)", orderId, did));
        }
        
        return order;
    }
    
    /**
     * 判断是否可以取消
     */
    private boolean canCancel(Integer progress) {
        return progress != null && 
               (progress == ServiceOrder.PROGRESS_ACCEPTED ||
                progress == ServiceOrder.PROGRESS_DISPATCHED ||
                progress == ServiceOrder.PROGRESS_RECEIVED ||
                progress == ServiceOrder.PROGRESS_ONSITE);
    }
   
    /**
     * 构建取消操作历史记录
     */
    private ServiceOrderHistory buildCancelHistory(ServiceOrder order, Integer lastStatus, 
                                                  String result, Long did, UserToken user) {
        return ServiceOrderHistory.builder()
                .serviceOrderId(order.getId())
                .operatorType(1) // 服务商操作
                .operationType(3) // 取消操作
                .operationContent(result != null ? result : "服务商取消")
                .lastStatus(lastStatus)
                .currentStatus(ServiceOrder.PROGRESS_CANCELED)
                .operatorId(did)
                .creatorId(user == null ? null : user.getId())
                .creatorName(user == null ? null : user.getName())
                .modifierId(user == null ? null : user.getId())
                .modifierName(user == null ? null : user.getName())
                .build();
    }
    
    /**
     * 构建接受操作历史记录
     */
    private ServiceOrderHistory buildAcceptHistory(ServiceOrder order, Long did, UserToken user) {
        return ServiceOrderHistory.builder()
                .serviceOrderId(order.getId())
                .operatorType(1) // 服务商操作
                .operationType(1) // 接受操作
                .operationContent("服务商接受服务单")
                .lastStatus(ServiceOrder.PROGRESS_WAIT_ACCEPT)
                .currentStatus(ServiceOrder.PROGRESS_ACCEPTED)
                .operatorId(did)
                .creatorId(user == null ? null : user.getId())
                .creatorName(user == null ? null : user.getName())
                .modifierId(user == null ? null : user.getId())
                .modifierName(user == null ? null : user.getName())
                .build();
    }
}

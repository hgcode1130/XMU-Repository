package cn.edu.xmu.oomall.service.dao.bo;

import cn.edu.xmu.javaee.core.exception.BusinessException;
import cn.edu.xmu.javaee.core.model.ReturnNo;
import cn.edu.xmu.javaee.core.model.UserToken;
import cn.edu.xmu.oomall.service.adapter.controller.dto.ConsigneeDto;
import cn.edu.xmu.oomall.service.adapter.controller.dto.CreateServiceOrderDto;
import cn.edu.xmu.oomall.service.model.ServiceBizConstants;
import jakarta.persistence.*;
import lombok.*;
import org.hibernate.annotations.CreationTimestamp;
import org.hibernate.annotations.UpdateTimestamp;
import org.springframework.util.StringUtils;

import java.time.LocalDateTime;

/**
 * 服务单实体
 */
@Entity
@Table(name = "service_order", uniqueConstraints = {
        @UniqueConstraint(name = "uk_service_order_shop_aftersale", columnNames = { "shop_id", "aftersale_id" })
})
@Getter
@Setter
@Builder
@NoArgsConstructor
@AllArgsConstructor
public class ServiceOrder {

    public static final int PROGRESS_WAIT_ACCEPT = ServiceBizConstants.OrderProgress.WAIT_ACCEPT;
    public static final int PROGRESS_ACCEPTED = ServiceBizConstants.OrderProgress.ACCEPTED;
    public static final int PROGRESS_DISPATCHED = ServiceBizConstants.OrderProgress.DISPATCHED;
    public static final int PROGRESS_RECEIVED = ServiceBizConstants.OrderProgress.RECEIVED;
    public static final int PROGRESS_ONSITE = ServiceBizConstants.OrderProgress.ONSITE;
    public static final int PROGRESS_FINISHED = ServiceBizConstants.OrderProgress.FINISHED;
    public static final int PROGRESS_RETURNED = ServiceBizConstants.OrderProgress.RETURNED;
    public static final int PROGRESS_CANCELED = ServiceBizConstants.OrderProgress.CANCELED;

    public static final int TYPE_ONSITE = 0;
    public static final int TYPE_MAIL = 1;
    public static final int TYPE_OFFLINE = 2;

    private static final int OPERATOR_TYPE_PROVIDER = ServiceBizConstants.OperatorType.PROVIDER;
    private static final int OPERATOR_TYPE_SYSTEM = ServiceBizConstants.OperatorType.SYSTEM;
    private static final int OPERATION_TYPE_CREATE = ServiceBizConstants.OrderOperationType.CREATE;
    private static final int OPERATION_TYPE_CANCEL = ServiceBizConstants.OrderOperationType.CANCEL;

    @Id
    @GeneratedValue(strategy = GenerationType.IDENTITY)
    private Long id;

    private Long aftersaleId;
    private Long shopId;
    private Integer type;
    private Long regionId;
    private String address;
    private String contact;
    @Column(name = "c_phone")
    private String phone;
    private Integer progress;
    private String maintainerName;
    private String maintainerMobile;
    private String description;
    private LocalDateTime reserveDate;
    private Long technicianId;
    @Column(name = "c_model")
    private String model;
    private LocalDateTime reserveTime;
    private Long serviceContractId;
    private String evidence;
    private Long serviceId;
    private Long reasonId;
    private String waybillNo;
    private Long creatorId;
    private String creatorName;
    private Long modifierId;
    private String modifierName;
    @CreationTimestamp
    private LocalDateTime gmtCreate;
    @UpdateTimestamp
    private LocalDateTime gmtModified;

    /**
     * 创建服务单初始化，业务校验下沉 BO
     * 引入策略模式处理不同类型的初始化差异
     */
    public void initForCreate(Long shopId, Long aftersaleId, CreateServiceOrderDto dto, UserToken user) {
        if (dto == null) {
            throw new BusinessException(ReturnNo.FIELD_NOTVALID, "请求体不能为空");
        }

        ServiceOrderType strategy = ServiceOrderType.fromCode(dto.getType());

        ConsigneeDto consignee = dto.getConsignee();
        if (consignee == null) {
            throw new BusinessException(ReturnNo.FIELD_NOTVALID, "consignee不能为空");
        }
        this.shopId = shopId;
        this.aftersaleId = aftersaleId;
        this.type = dto.getType();
        this.regionId = consignee.getRegionId();
        this.address = consignee.getAddress();
        this.contact = consignee.getName();
        this.phone = consignee.getMobile();
        if (dto.getServiceContractId() != null && dto.getServiceContractId() <= 0) {
            throw new BusinessException(ReturnNo.FIELD_NOTVALID, "serviceContractId非法");
        }
        this.serviceContractId = dto.getServiceContractId();

        // 委托策略进行初始化
        strategy.init(this);

        this.creatorId = user == null ? null : user.getId();
        this.creatorName = user == null ? null : user.getName();
    }

    /**
     * 服务单类型策略枚举
     * 封装不同类型的初始化、取消等差异化逻辑
     */
    @AllArgsConstructor
    @Getter
    private enum ServiceOrderType {
        ONSITE(TYPE_ONSITE) {
            @Override
            public void init(ServiceOrder order) {
                // 上门服务单初始状态为待接受（待网点/师傅接单）
                order.setProgress(PROGRESS_WAIT_ACCEPT);
            }
        },
        MAIL(TYPE_MAIL) {
            @Override
            public void init(ServiceOrder order) {
                // 寄修服务单初始状态也为待接受（待网点确认接收寄修请求）
                // 未来如有不同逻辑（如直接生成运单）可在此扩展
                order.setProgress(PROGRESS_WAIT_ACCEPT);
            }
        },
        OFFLINE(TYPE_OFFLINE) {
            @Override
            public void init(ServiceOrder order) {
                order.setProgress(PROGRESS_WAIT_ACCEPT);
            }
        };

        private final int code;

        public abstract void init(ServiceOrder order);

        public static ServiceOrderType fromCode(Integer code) {
            if (code == null)
                throw new BusinessException(ReturnNo.FIELD_NOTVALID, "type不能为空");
            for (ServiceOrderType type : values()) {
                if (type.code == code) {
                    return type;
                }
            }
            throw new BusinessException(ReturnNo.FIELD_NOTVALID, "无效的服务单类型: " + code);
        }
    }

    /**
     * 生成创建历史记录
     */
    public ServiceOrderHistory buildCreationHistory(UserToken user) {
        return ServiceOrderHistory.builder()
                .serviceOrderId(this.id)
                .operatorType(OPERATOR_TYPE_SYSTEM)
                .operationType(OPERATION_TYPE_CREATE)
                .operationContent("售后创建服务单")
                .lastStatus(null)
                .currentStatus(this.progress)
                .operatorId(user == null ? null : user.getId())
                .creatorId(user == null ? null : user.getId())
                .creatorName(user == null ? null : user.getName())
                .build();
    }

    /**
     * 服务商取消服务单，校验归属与进度
     */
    public ServiceOrderHistory cancel(ServiceContract contract, Long did, String result, UserToken user) {
        if (serviceContractId == null || contract == null || !serviceContractId.equals(contract.getId())) {
            throw new BusinessException(ReturnNo.RESOURCE_ID_OUTSCOPE,
                    String.format("服务单(id=%d)未绑定当前服务商合同", id));
        }
        if (!contract.ownedBy(did)) {
            throw new BusinessException(ReturnNo.RESOURCE_ID_OUTSCOPE,
                    String.format("服务单(id=%d)不属于服务商(%d)", id, did));
        }
        if (!canCancel()) {
            throw new BusinessException(ReturnNo.STATENOTALLOW,
                    String.format("服务单(id=%d)状态%d不允许取消", id, progress));
        }
        int lastStatus = progress == null ? 0 : progress;
        this.progress = PROGRESS_CANCELED;
        applyModifier(user);
        String content = StringUtils.hasText(result) ? result : "服务商取消";

        return ServiceOrderHistory.builder()
                .serviceOrderId(this.id)
                .operatorType(OPERATOR_TYPE_PROVIDER)
                .operationType(OPERATION_TYPE_CANCEL)
                .operationContent(content)
                .lastStatus(lastStatus)
                .currentStatus(PROGRESS_CANCELED)
                .operatorId(did)
                .creatorId(user == null ? null : user.getId())
                .creatorName(user == null ? null : user.getName())
                .modifierId(user == null ? null : user.getId())
                .modifierName(user == null ? null : user.getName())
                .build();
    }

    private boolean canCancel() {
        return progress != null && ServiceBizConstants.OrderProgress.CANCEL_ALLOWED.contains(progress);
    }

    private void applyModifier(UserToken user) {
        this.modifierId = user == null ? null : user.getId();
        this.modifierName = user == null ? null : user.getName();
    }
}

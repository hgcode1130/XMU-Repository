package cn.edu.xmu.oomall.service.dao.bo;

import cn.edu.xmu.javaee.core.exception.BusinessException;
import cn.edu.xmu.javaee.core.model.ReturnNo;
import cn.edu.xmu.javaee.core.model.UserToken;
import cn.edu.xmu.oomall.service.model.ServiceBizConstants;
import jakarta.persistence.*;
import lombok.*;
import org.hibernate.annotations.CreationTimestamp;
import org.hibernate.annotations.UpdateTimestamp;
import org.springframework.util.StringUtils;

import java.time.LocalDateTime;
import java.util.Objects;

/**
 * 服务合同实体，负责状态校验与取消逻辑
 */
@Entity
@Table(name = "service_contract")
@Getter
@Setter
@Builder
@NoArgsConstructor
@AllArgsConstructor
public class ServiceContract {

    public static final int STATUS_APPLY = ServiceBizConstants.ContractStatus.APPLY;
    public static final int STATUS_ACTIVE = ServiceBizConstants.ContractStatus.ACTIVE;
    public static final int STATUS_PAUSED = ServiceBizConstants.ContractStatus.PAUSED;
    public static final int STATUS_CANCELED = ServiceBizConstants.ContractStatus.CANCELED;
    public static final int STATUS_REJECTED = ServiceBizConstants.ContractStatus.REJECTED;
    public static final int STATUS_EXPIRED = ServiceBizConstants.ContractStatus.EXPIRED;

    private static final int OPERATOR_TYPE_PROVIDER = ServiceBizConstants.OperatorType.PROVIDER;
    private static final int OPERATION_TYPE_CANCEL = ServiceBizConstants.ContractOperationType.CANCEL;

    @Id
    @GeneratedValue(strategy = GenerationType.IDENTITY)
    private Long id;

    @Column(name = "service_id")
    private Long serviceId;

    @Column(name = "service_provider_id")
    private Long serviceProviderId;

    private Integer status;

    @Column(name = "region_id")
    private Long regionId;

    private String name;

    private String description;

    @Column(name = "start_at")
    private LocalDateTime startAt;

    @Column(name = "end_at")
    private LocalDateTime endAt;

    @Column(name = "service_order_id")
    private Long serviceOrderId;

    @Column(name = "creator_id")
    private Long creatorId;

    @Column(name = "creator_name")
    private String creatorName;

    @Column(name = "modifier_id")
    private Long modifierId;

    @Column(name = "modifier_name")
    private String modifierName;

    @CreationTimestamp
    @Column(name = "gmt_create")
    private LocalDateTime gmtCreate;

    @UpdateTimestamp
    @Column(name = "gmt_modified")
    private LocalDateTime gmtModified;

    /**
     * 服务商取消合同
     */
    public ContractHistory cancel(Long did, String reason, UserToken user) {
        if (!ownedBy(did)) {
            throw new BusinessException(ReturnNo.RESOURCE_ID_OUTSCOPE,
                    String.format("合同(id=%d)不属于服务商(%d)", id, did));
        }
        if (!canCancel()) {
            throw new BusinessException(ReturnNo.STATENOTALLOW,
                    String.format("合同(id=%d)状态%d不允许取消", id, status));
        }
        this.status = STATUS_CANCELED;
        applyModifier(user);
        this.gmtModified = LocalDateTime.now();
        String finalReason = StringUtils.hasText(reason) ? reason : "服务商取消";
        return ContractHistory.builder()
                .serviceContractId(this.id)
                .operatorType(OPERATOR_TYPE_PROVIDER)
                .operationType(OPERATION_TYPE_CANCEL)
                .operationTime(LocalDateTime.now())
                .reason(finalReason)
                .operatorId(did)
                .creatorId(user == null ? null : user.getId())
                .creatorName(user == null ? null : user.getName())
                .modifierId(user == null ? null : user.getId())
                .modifierName(user == null ? null : user.getName())
                .build();
    }

    public boolean ownedBy(Long did) {
        return serviceProviderId != null && Objects.equals(serviceProviderId, did);
    }

    public String statusText() {
        return switch (status == null ? -1 : status) {
            case STATUS_APPLY -> "申请中";
            case STATUS_ACTIVE -> "生效";
            case STATUS_PAUSED -> "暂停";
            case STATUS_CANCELED -> "取消";
            case STATUS_REJECTED -> "拒绝";
            case STATUS_EXPIRED -> "到期";
            default -> "未知状态";
        };
    }

    private boolean canCancel() {
        return status != null && (status == STATUS_APPLY || status == STATUS_ACTIVE || status == STATUS_PAUSED);
    }

    private void applyModifier(UserToken user) {
        this.modifierId = user == null ? null : user.getId();
        this.modifierName = user == null ? null : user.getName();
    }
}

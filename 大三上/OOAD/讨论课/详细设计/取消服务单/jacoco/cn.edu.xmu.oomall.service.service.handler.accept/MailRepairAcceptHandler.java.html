package cn.edu.xmu.oomall.service.service.handler.accept;

import cn.edu.xmu.oomall.service.adapter.feign.LogisticsService;
import cn.edu.xmu.oomall.service.adapter.feign.dto.CreateWaybillRequest;
import cn.edu.xmu.oomall.service.adapter.feign.dto.WaybillResponse;
import cn.edu.xmu.oomall.service.dao.bo.ServiceOrder;
import cn.edu.xmu.oomall.service.service.handler.AcceptHandler;
import cn.edu.xmu.oomall.service.service.result.AcceptResult;
import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.stereotype.Component;

import java.time.LocalDateTime;
import java.time.format.DateTimeFormatter;

/**
 * 寄修服务接受处理器
 * 场景：寄修服务需要生成运单让客户寄送物品
 */
@Component
@RequiredArgsConstructor
@Slf4j
public class MailRepairAcceptHandler implements AcceptHandler {
    
    private static final DateTimeFormatter FORMATTER = DateTimeFormatter.ofPattern("yyyyMMddHHmmss");
    
    private final LogisticsService logisticsService;
    
    @Override
    public boolean supports(Integer serviceType) {
        return ServiceOrder.TYPE_MAIL == serviceType;
    }
    
    @Override
    public AcceptResult accept(ServiceOrder serviceOrder, Long providerId) {
        log.info("开始处理寄修服务接受 | orderId={} providerId={}", 
                serviceOrder.getId(), providerId);
        
        // 构建运单请求
        CreateWaybillRequest request = CreateWaybillRequest.builder()
                .receiverName("维修中心")
                .receiverPhone("400-000-0000")
                .receiverRegionId(1L)
                .receiverAddress("维修中心地址")
                .senderName(serviceOrder.getContact())
                .senderPhone(serviceOrder.getPhone())
                .senderRegionId(serviceOrder.getRegionId())
                .senderAddress(serviceOrder.getAddress())
                .build();
        
        String waybillNo;
        
        try {
            // 尝试调用物流服务创建运单
            var response = logisticsService.createWaybill(request);
            if (response != null && response.getData() != null) {
                WaybillResponse waybillResponse = (WaybillResponse) response.getData();
                waybillNo = waybillResponse.getWaybillNo();
                log.info("✅ 成功调用物流服务创建运单 | waybillNo={}", waybillNo);
            } else {
                // 物流服务返回空数据，使用降级处理
                waybillNo = generateMockWaybillNo();
                log.warn("⚠️ 物流服务返回空数据，使用模拟运单号 | waybillNo={}", waybillNo);
            }
        } catch (Exception e) {
            // 物流服务调用失败，使用降级处理
            waybillNo = generateMockWaybillNo();
            log.warn("⚠️ 物流服务调用失败，使用降级处理 | error={} | 生成模拟运单号={}", 
                    e.getMessage(), waybillNo);
            log.info("📦 模拟运单信息 | 寄件人={} 电话={} 地址={}", 
                    request.getSenderName(), 
                    request.getSenderPhone(), 
                    request.getSenderAddress());
        }
        
        // 更新服务单状态和运单号
        serviceOrder.setProgress(ServiceOrder.PROGRESS_ACCEPTED);
        serviceOrder.setWaybillNo(waybillNo);
        
        log.info("寄修服务接受完成 | orderId={} waybillNo={}", serviceOrder.getId(), waybillNo);
        return AcceptResult.success("接受成功，已生成运单", waybillNo);
    }
    
    /**
     * 生成模拟运单号（降级处理）
     */
    private String generateMockWaybillNo() {
        return "MOCK-" + LocalDateTime.now().format(FORMATTER);
    }
}

package cn.edu.xmu.oomall.service.service.handler.accept;

import cn.edu.xmu.oomall.service.dao.bo.ServiceOrder;
import cn.edu.xmu.oomall.service.service.handler.AcceptHandler;
import cn.edu.xmu.oomall.service.service.result.AcceptResult;
import lombok.extern.slf4j.Slf4j;
import org.springframework.stereotype.Component;

/**
 * 上门服务接受处理器
 * 场景：上门服务仅需更新状态，等待后续派工
 */
@Component
@Slf4j
public class OnsiteAcceptHandler implements AcceptHandler {
    
    @Override
    public boolean supports(Integer serviceType) {
        return ServiceOrder.TYPE_ONSITE == serviceType;
    }
    
    @Override
    public AcceptResult accept(ServiceOrder serviceOrder, Long providerId) {
        log.info("开始处理上门服务接受 | orderId={} providerId={}", 
                serviceOrder.getId(), providerId);
        
        // 仅更新服务单状态
        serviceOrder.setProgress(ServiceOrder.PROGRESS_ACCEPTED);
        
        return AcceptResult.success("接受成功，等待派工");
    }
}

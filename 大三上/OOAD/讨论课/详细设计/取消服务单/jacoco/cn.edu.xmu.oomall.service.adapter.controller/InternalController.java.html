package cn.edu.xmu.oomall.service.adapter.controller;

import cn.edu.xmu.javaee.core.aop.Audit;
import cn.edu.xmu.javaee.core.aop.UserLevel;
import cn.edu.xmu.javaee.core.model.ReturnNo;
import cn.edu.xmu.javaee.core.model.ReturnObject;
import cn.edu.xmu.javaee.core.model.UserToken;
import cn.edu.xmu.oomall.service.adapter.controller.dto.CreateServiceOrderDto;
import cn.edu.xmu.oomall.service.adapter.controller.vo.SimpleServiceVo;
import cn.edu.xmu.oomall.service.dao.bo.ServiceOrder;
import cn.edu.xmu.oomall.service.service.ServiceOrderService;
import jakarta.validation.Valid;
import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.transaction.annotation.Propagation;
import org.springframework.transaction.annotation.Transactional;
import org.springframework.validation.annotation.Validated;
import org.springframework.web.bind.annotation.*;

/**
 * 内部接口：创建服务单
 */
@RestController
@RequestMapping(value = "/internal", produces = "application/json;charset=UTF-8")
@Slf4j
@RequiredArgsConstructor
@Validated
public class InternalController {

    private final ServiceOrderService serviceOrderService;

    @PostMapping("/shops/{shopId}/aftersales/{id}/serviceorders")
    @Audit(departName = "shops")
    @Transactional(propagation = Propagation.REQUIRED)
    public ReturnObject createServiceOrder(@PathVariable Long shopId,
                                           @PathVariable("id") Long aftersaleId,
                                           @Valid @RequestBody CreateServiceOrderDto dto,
                                           @cn.edu.xmu.javaee.core.aop.LoginUser UserToken user,
                                           @UserLevel Integer userLevel) {
        log.info("create service order request | shopId={} aftersaleId={} type={} consignee={} contractId={}",
                shopId, aftersaleId, dto.getType(), dto.getConsignee().getName(), dto.getServiceContractId());
        ReturnObject ret = serviceOrderService.createServiceOrder(shopId, aftersaleId, dto, user);
        if (ret.getErrno() != ReturnNo.OK.getErrNo() && ret.getErrno() != ReturnNo.CREATED.getErrNo()) {
            return ret;
        }
        ServiceOrder order = (ServiceOrder) ret.getData();
        SimpleServiceVo vo = new SimpleServiceVo();
        vo.setId(order.getId());
        vo.setType(order.getType());
        vo.setConsignee(order.getContact());
        ReturnNo code = ret.getErrno() == ReturnNo.CREATED.getErrNo() ? ReturnNo.CREATED : ReturnNo.OK;
        return new ReturnObject(code, vo);
    }
}

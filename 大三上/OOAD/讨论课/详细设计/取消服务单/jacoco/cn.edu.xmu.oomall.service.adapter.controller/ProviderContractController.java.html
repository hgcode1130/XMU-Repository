package cn.edu.xmu.oomall.service.adapter.controller;

import cn.edu.xmu.javaee.core.aop.Audit;
import cn.edu.xmu.javaee.core.aop.UserLevel;
import cn.edu.xmu.javaee.core.model.ReturnNo;
import cn.edu.xmu.javaee.core.model.ReturnObject;
import cn.edu.xmu.javaee.core.model.UserToken;
import cn.edu.xmu.oomall.service.adapter.controller.dto.CancelContractDto;
import cn.edu.xmu.oomall.service.adapter.controller.vo.ContractStatusVo;
import cn.edu.xmu.oomall.service.dao.bo.ServiceContract;
import cn.edu.xmu.oomall.service.service.ServiceContractService;
import cn.edu.xmu.oomall.service.service.ServiceContractService.CancelContractResult;
import jakarta.validation.Valid;
import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.transaction.annotation.Propagation;
import org.springframework.transaction.annotation.Transactional;
import org.springframework.validation.annotation.Validated;
import org.springframework.web.bind.annotation.*;

/**
 * 服务商合同接口
 */
@RestController
@RequestMapping(value = "/providers/contracts", produces = "application/json;charset=UTF-8")
@Slf4j
@Validated
@RequiredArgsConstructor
public class ProviderContractController {

    private final ServiceContractService serviceContractService;

    @PutMapping("/{id}/cancel")
    @Audit(departName = "providers")
    @Transactional(propagation = Propagation.REQUIRED)
    public ReturnObject cancel(@PathVariable("id") Long contractId,
                               @Valid @RequestBody(required = false) CancelContractDto dto,
                               @cn.edu.xmu.javaee.core.aop.LoginUser UserToken user,
                               @UserLevel Integer userLevel) {
        Long did = user == null ? null : user.getDepartId();
        String reason = dto == null ? null : dto.getReason();
        log.info("provider cancel contract request | did={} contractId={} reason={}", did, contractId, reason);

        ReturnObject ret = serviceContractService.cancel(did, contractId, reason, user);
        if (ret.getErrno() != ReturnNo.OK.getErrNo()) {
            return ret;
        }
        CancelContractResult result = (CancelContractResult) ret.getData();
        ServiceContract contract = result.contract();
        ContractStatusVo vo = new ContractStatusVo();
        vo.setId(contract.getId());
        vo.setStatus(contract.getStatus());
        vo.setStatusText(contract.statusText());
        vo.setReason(result.reason());
        vo.setUpdatedAt(contract.getGmtModified());
        return new ReturnObject(ReturnNo.OK, vo);
    }
}

package cn.edu.xmu.oomall.service.adapter.controller;

import cn.edu.xmu.javaee.core.aop.Audit;
import cn.edu.xmu.javaee.core.aop.UserLevel;
import cn.edu.xmu.javaee.core.model.ReturnObject;
import cn.edu.xmu.javaee.core.model.UserToken;
import cn.edu.xmu.oomall.service.adapter.controller.dto.CancelServiceOrderDto;
import cn.edu.xmu.oomall.service.service.ProviderServiceOrderService;
import jakarta.validation.Valid;
import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.transaction.annotation.Propagation;
import org.springframework.transaction.annotation.Transactional;
import org.springframework.validation.annotation.Validated;
import org.springframework.web.bind.annotation.*;

/**
 * 服务商侧服务单接口
 */
@RestController
@RequestMapping(value = "/providers/{did}/service", produces = "application/json;charset=UTF-8")
@Slf4j
@Validated
@RequiredArgsConstructor
public class ProviderServiceOrderController {

    private final ProviderServiceOrderService providerServiceOrderService;

    @PutMapping("/{id}/cancel")
    @Audit(departName = "providers")
    @Transactional(propagation = Propagation.REQUIRED)
    public ReturnObject cancel(@PathVariable Long did,
                               @PathVariable("id") Long orderId,
                               @Valid @RequestBody(required = false) CancelServiceOrderDto dto,
                               @cn.edu.xmu.javaee.core.aop.LoginUser UserToken user,
                               @UserLevel Integer userLevel) {
        String result = dto == null ? null : dto.getResult();
        log.info("provider cancel service order request | did={} orderId={} result={}", did, orderId, result);
        return providerServiceOrderService.cancel(did, orderId, result, user);
    }

    @PutMapping("/{id}/accept")
    @Audit(departName = "providers")
    @Transactional(propagation = Propagation.REQUIRED)
    public ReturnObject accept(@PathVariable Long did,
                              @PathVariable("id") Long orderId,
                              @cn.edu.xmu.javaee.core.aop.LoginUser UserToken user,
                              @UserLevel Integer userLevel) {
        log.info("provider accept service order request | did={} orderId={}", did, orderId);
        return providerServiceOrderService.accept(did, orderId, user);
    }
}

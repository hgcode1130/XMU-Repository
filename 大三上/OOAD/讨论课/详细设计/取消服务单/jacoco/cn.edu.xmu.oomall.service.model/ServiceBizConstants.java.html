package cn.edu.xmu.oomall.service.model;

import java.util.Set;

/**
 * 服务领域的状态与操作常量，集中管理以便维护与扩展
 */
public final class ServiceBizConstants {

    private ServiceBizConstants() {
    }

    public static final class OrderProgress {
        public static final int WAIT_ACCEPT = 0;
        public static final int ACCEPTED = 1;
        public static final int DISPATCHED = 2;
        public static final int RECEIVED = 3;
        public static final int ONSITE = 4;
        public static final int FINISHED = 5;
        public static final int RETURNED = 6;
        public static final int CANCELED = 7;

        public static final Set<Integer> CANCEL_ALLOWED = Set.of(ACCEPTED, DISPATCHED, RECEIVED, ONSITE);
    }

    public static final class OrderOperationType {
        public static final int CREATE = 0;
        public static final int CANCEL = 3;
    }

    public static final class ContractStatus {
        public static final int APPLY = 0;
        public static final int ACTIVE = 1;
        public static final int PAUSED = 2;
        public static final int CANCELED = 3;
        public static final int REJECTED = 4;
        public static final int EXPIRED = 5;
    }

    public static final class ContractOperationType {
        public static final int CANCEL = 3;
    }

    public static final class OperatorType {
        public static final int PROVIDER = 1;
        public static final int SYSTEM = 2;
    }
}
